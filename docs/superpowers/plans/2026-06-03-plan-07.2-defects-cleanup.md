| # | Decision | Rationale |
|---|---|---|
| D1 | Template detection: composite signals (3 cells minimum) instead of single A2 check | Reduce false-positive misclassification |
| D2 | `ImportBatch.SourceFileHash` gets non-unique index + de-dup warning at save time | Detect duplicate imports without blocking — let user confirm |
| D3 | Auto-navigate to TongHop moved out of MainWindowViewModel constructor → `InitializeAsync()` called from `App.axaml.cs` after host built | Exceptions observable; testable |
| D4 | `[Auditable]` attribute NOT added to ImportBatch / ProjectSummary | Import is a bulk op; per-row audit pollutes audit log. Existing batch metadata sufficient |
| D5 | Year regex tightened to `Quyết\s*toán\s*(?:đến\s*)?năm\s+(\d{4})\b` (boundary) | Avoid 5+ digit year false matches |
| D6 | LINQ side-effect (warnings) refactored to explicit `foreach` | Lint-friendly, no behavior change |
| D7 | Hieu.xlsx fixture: rename source path to ASCII `Hieu_sample.xlsx` (copy from `reference/Hiệu.xlsx`) | Avoid Windows MSBuild encoding issues |

### Task 2: I4 — Template detection by composite signals

**File:** `src/ConstructionFinance.Application/Services/Implementations/ExcelImportService.cs`

#### Step 2.1: Refactor `DetectDaTemplate`

```csharp
private static DaTemplate DetectDaTemplate(IXLWorksheet sheet, int totalRowIdx)
{
    // Score-based detection: collect signals, count Hieu vs H matches.
    int hieuScore = 0, hScore = 0;
    var a2 = sheet.Cell(2, 1).GetString().Trim();
    var b4 = sheet.Cell(4, 2).GetString().Trim();
    var c2 = sheet.Cell(2, 3).GetString().Trim();
    var b2 = sheet.Cell(2, 2).GetString().Trim();

    if (a2.Contains("Ngày tháng", StringComparison.OrdinalIgnoreCase)) hieuScore++;
    if (b2.Contains("Tên gói thầu", StringComparison.OrdinalIgnoreCase)) hieuScore++;
    if (c2.Contains("Hợp đồng", StringComparison.OrdinalIgnoreCase)) hieuScore++;
    if (totalRowIdx == 4) hieuScore++;

    var a4 = sheet.Cell(4, 1).GetString().Trim();
    var b4Main = sheet.Cell(4, 2).GetString().Trim();
    if (a4.Equals("TT", StringComparison.OrdinalIgnoreCase)) hScore++;
    if (b4Main.Contains("Nội dung", StringComparison.OrdinalIgnoreCase)) hScore++;
    if (totalRowIdx == 6) hScore++;

    return hieuScore > hScore ? DaTemplate.Hieu_xlsx : DaTemplate.H_xlsx;
}
```

#### Step 4.1: M1 — Refactor `RefineFundingFromTongHop` return

Replace `.Select(s => { ... warnings.Add(...) ... return s; }).ToList()` with:

```csharp
var refined = new List<ProjectSummaryDto>(summaries.Count);
foreach (var s in summaries)
{
    var updated = TryRefine(s, bySheetRef, byName);
    if (updated is null && s.TotalDisbursed > 0)
        warnings.Add($"Dự án '{s.ProjectName}' (sheet '{s.SheetName}'): không tìm thấy phân tách NSNN/DTNH.");
    refined.Add(updated ?? s);
}
return refined;

static ProjectSummaryDto? TryRefine(
    ProjectSummaryDto s,
    Dictionary<string, (decimal nsnn, decimal dtnh)> bySheetRef,
    Dictionary<string, (decimal nsnn, decimal dtnh)> byName)
{
    if (s.DisbursedDTNH != 0m || s.DisbursedNSNN != s.TotalDisbursed) return null;
    if (bySheetRef.TryGetValue(s.SheetName, out var a) && a.nsnn + a.dtnh > 0)
        return s with { DisbursedNSNN = a.nsnn, DisbursedDTNH = a.dtnh };
    if (byName.TryGetValue(s.ProjectName, out var b) && b.nsnn + b.dtnh > 0)
        return s with { DisbursedNSNN = b.nsnn, DisbursedDTNH = b.dtnh };
    return null;
}
```

### Task 5: M3 — Safe WarningsJson deserialization

**File:** `src/ConstructionFinance.Application/Mapping/ProjectSummaryMappings.cs`

```csharp
public static ImportBatchDto ToDto(this ImportBatch b)
{
    IReadOnlyList<string> warnings = Array.Empty<string>();
    if (!string.IsNullOrEmpty(b.WarningsJson))
    {
        try { warnings = JsonSerializer.Deserialize<List<string>>(b.WarningsJson) ?? Array.Empty<string>(); }
        catch (JsonException) { warnings = new[] { "(Cảnh báo bị hỏng JSON.)" }; }
    }
    return new ImportBatchDto(b.Id, b.SourceFileName, b.ImportedAtUtc,
        b.SheetsFound, b.ProjectsImported, b.Description, warnings);
}
```
